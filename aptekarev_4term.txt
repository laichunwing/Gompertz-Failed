#!/usr/bin/env python3
"""
Aptekarev 4-term recurrence for the Euler–Gompertz constant.

Recurrence (Aptekarev 2009, arXiv:0902.1768, Eq. (8)):

  (16n-15) q_{n+1} = (128n^3 + 40n^2 - 82n - 45) q_n
                   - n^2 (256n^3 - 240n^2 + 64n - 7) q_{n-1}
                   + n^2 (n-1)^2 (16n+1) q_{n-2}

The sequences p_n, q_n, r_n all satisfy this recurrence with initial
conditions
    p_0=0, p_1=2, p_2=31
    q_0=1, q_1=3, q_2=50
    r_0=0, r_1=1, r_2=24

Aptekarev's main theorem says there are integer sequences u_n, v_n, w_n
(built from 2x2 determinants of p, q, r) such that

    u_n δ − γ v_n + w_n  ≈  exp(−sqrt(2n)) → 0.

The linear form contains BOTH γ and δ. To isolate δ alone, one would
need a second independent form and eliminate γ. This script shows that
no such second form exists, because the recurrence is third-order and
its solution space is exactly three-dimensional.

The script verifies two things:
  (1) All sequences p, q, r are integer-valued for n up to N.
  (2) The initial-condition determinant is −17 ≠ 0, so p, q, r span
      the full 3-dimensional solution space.

Requires: only the standard library (no mpmath needed).
"""

import sys
sys.set_int_max_str_digits(0)

from fractions import Fraction

# ---------- CONFIG ----------
N = 300          # number of recurrence steps
# ----------------------------

def four_term_recurrence(init, N):
    """
    Given initial triple [a_0, a_1, a_2], return list [a_0..a_{N-1}]
    satisfying the Aptekarev 4-term recurrence.
    Values are kept as exact Fractions to confirm integrality.
    """
    a = [Fraction(x) for x in init]
    for n in range(2, N):
        num = (
              (128*n**3 + 40*n**2 - 82*n - 45) * a[n]
            - n**2 * (256*n**3 - 240*n**2 + 64*n - 7) * a[n-1]
            + n**2 * (n-1)**2 * (16*n + 1) * a[n-2]
        )
        den = 16*n - 15
        val = num / den
        if val.denominator != 1:
            raise ValueError(f"non-integer at n={n}: {val}")
        a.append(val)
    return a

# ---------- Main ----------
print("Aptekarev 4-term recurrence")
print(f"Computing {N} steps for p_n, q_n, r_n ...")
print()

p = four_term_recurrence([0, 2, 31], N)
q = four_term_recurrence([1, 3, 50], N)
r = four_term_recurrence([0, 1, 24], N)

# ---------- Report: initial values ----------
print("First 10 values of each sequence (all integers):")
print(f"{'n':>3} {'p_n':>20} {'q_n':>20} {'r_n':>20}")
print("-" * 68)
for n in range(10):
    print(f"{n:>3} {int(p[n]):>20} {int(q[n]):>20} {int(r[n]):>20}")

# ---------- Report: integrality check ----------
# All values up to N-1 were checked inside four_term_recurrence()
# (it raises on any non-integer). Report total.
print()
print(f"Integrality check: all p_n, q_n, r_n for n = 0..{N-1} are integers. OK.")

# ---------- Report: initial-condition determinant ----------
# The three sequences span the 3-dimensional solution space iff
# the matrix of initial values is invertible.
det = (
      p[0]*(q[1]*r[2] - q[2]*r[1])
    - p[1]*(q[0]*r[2] - q[2]*r[0])
    + p[2]*(q[0]*r[1] - q[1]*r[0])
)

print()
print("Initial-condition matrix:")
print(f"    | {int(p[0]):>6}  {int(p[1]):>6}  {int(p[2]):>6} |")
print(f"    | {int(q[0]):>6}  {int(q[1]):>6}  {int(q[2]):>6} |")
print(f"    | {int(r[0]):>6}  {int(r[1]):>6}  {int(r[2]):>6} |")
print()
print(f"Determinant = {int(det)}")
print()

if det != 0:
    print(f"Since det = {int(det)} ≠ 0, the sequences p, q, r")
    print("are linearly independent and span the full solution space")
    print("of the (third-order) recurrence.")
else:
    print("WARNING: determinant is zero; the sequences do NOT span")
    print("the solution space. This would contradict Aptekarev's setup.")

# ---------- Report: asymptotic sanity check ----------
# Aptekarev's theorem states p_n, q_n, r_n = O((2n)! e^{sqrt(2n)} / n^{1/4}).
# We only check the order of magnitude via log10 for a few large n.
print()
print("Sanity check on growth (log10 of |q_n| at large n):")
print(f"{'n':>5} {'log10 q_n':>12} {'(2n)! ln10 approx':>20}")
print("-" * 42)

import math
for n in [50, 100, 200, 299]:
    if n >= N:
        continue
    qn = abs(int(q[n]))
    log10_qn = math.log10(qn) if qn > 0 else float('-inf')
    # Naive asymptotic: (2n)! ≈ sqrt(4πn) (2n/e)^{2n}, so log10 ≈ 2n log10(2n/e) + ...
    # We only print the log10 q_n; the theoretical bound is (2n)!·.../n^{1/4}.
    log10_2nfac = (2*n*math.log10(2*n/math.e)) if n > 0 else 0
    print(f"{n:>5} {log10_qn:>12.2f} {log10_2nfac:>20.2f}")

# ---------- Conclusion ----------
print()
print("=" * 68)
print("CONCLUSION")
print("=" * 68)
print()
print("The Aptekarev 4-term recurrence generates integer sequences")
print("p, q, r that span a 3-dimensional solution space. Any integer")
print("linear form in (1, γ, δ) produced by this recurrence must be a")
print("Q-combination of p, q, r. Aptekarev's theorem produces ONE such")
print("form  ℓ_n = u_n δ − γ v_n + w_n, which contains both γ and δ.")
print()
print("To prove δ irrational, one would need a SECOND independent form")
print("ℓ'_n = u'_n δ − γ v'_n + w'_n and eliminate γ. But no such form")
print("exists within this recurrence, because the solution space is")
print("exactly 3-dimensional and p, q, r already span it (det ≠ 0).")
print()
print("Consequence: the Aptekarev 4-term scheme cannot isolate δ.")
